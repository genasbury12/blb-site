<script>
(function(){
  var pop=document.getElementById('blbExit');
  if(!pop)return;
  var KEY='blbExitSeen';
  try{
    var seen=localStorage.getItem(KEY);
    if(seen==='grabbed')return;
    if(seen&&Date.now()-parseInt(seen,10)<7*24*60*60*1000)return;
  }catch(e){}
  var shown=false,armed=false;
  setTimeout(function(){armed=true},6000);
  function redesignOpen(){
    var rd=document.getElementById('blbPop');
    return rd&&getComputedStyle(rd).display!=='none';
  }
  function show(){
    if(shown||!armed||redesignOpen())return;
    shown=true;
    pop.style.display='flex';
  }
  function close(val){
    pop.style.display='none';
    try{localStorage.setItem(KEY,val||String(Date.now()))}catch(e){}
  }
  /* desktop: mouse bolts for the top of the window */
  document.addEventListener('mouseout',function(e){
    if(!e.relatedTarget&&e.clientY<=12)show();
  });
  /* phone: scrolled well in, then a fast flick back up */
  var maxY=0,upRun=0,lastY=window.pageYOffset;
  document.addEventListener('scroll',function(){
    var y=window.pageYOffset;
    if(y>maxY)maxY=y;
    if(y<lastY)upRun+=lastY-y;else upRun=0;
    lastY=y;
    if(maxY>700&&upRun>450)show();
  },{passive:true});
  pop.querySelectorAll('[data-close]').forEach(function(el){
    el.addEventListener('click',function(){close()});
  });
  document.addEventListener('keydown',function(e){
    if(e.key==='Escape'&&pop.style.display==='flex')close();
  });
  var btn=pop.querySelector('.btn');
  if(btn)btn.addEventListener('click',function(){close('grabbed')});
})();
</script>
