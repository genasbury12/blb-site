<script>
(function(){
  var pop=document.getElementById('blbPop');
  if(!pop)return;
  var KEY='blbPopSeen';
  try{
    var seen=localStorage.getItem(KEY);
    if(seen==='booked')return;
    if(seen&&Date.now()-parseInt(seen,10)<7*24*60*60*1000)return;
  }catch(e){}
  var shown=false;
  function show(){
    if(shown)return;shown=true;
    pop.style.display='flex';
    document.removeEventListener('scroll',onScroll);
  }
  function close(val){
    pop.style.display='none';
    try{localStorage.setItem(KEY,val||String(Date.now()))}catch(e){}
  }
  function onScroll(){
    var d=document.documentElement;
    var max=d.scrollHeight-d.clientHeight;
    if(max>0&&d.scrollTop/max>0.33)show();
  }
  setTimeout(show,9000);
  document.addEventListener('scroll',onScroll,{passive:true});
  pop.querySelectorAll('[data-close]').forEach(function(el){
    el.addEventListener('click',function(){close()});
  });
  document.addEventListener('keydown',function(e){
    if(e.key==='Escape'&&pop.style.display==='flex')close();
  });
  var btn=pop.querySelector('.btn');
  if(btn)btn.addEventListener('click',function(){close('booked')});
})();
