  WHAT THIS PAGE IS
  The page she opens most. Not a place to browse and admire, a place
  to find one page and print it in under ten seconds. Everything here
  is built for speed of retrieval, not for showing off the catalogue.

  HOW THE LIST IS BUILT
  The 27 printables live in the ITEMS array at the bottom of the file
  and the cards are rendered from it. To add a printable, add one
  object to that array. Do not hand write card markup, or the counts
  and filters will drift out of sync within a month.

  Each item needs:
     title, pages, blurb, cat, system, file
     isNew: true   puts a New flag on it and floats it under
                   the New this month filter
     starter: true adds it to the print these three first strip

  COUNTS MUST AGREE WITH THE VAULT
  Every vault card states a printable count. Those numbers come from
  this array. Current totals, checked by the console log at the
  bottom of the script:
     01 budget 4  ◆  02 sinking funds 3  ◆  03 meals 4
     04 groceries 4 ◆ 05 home 4        ◆  06 mindset 4
     07 implementation 2                ◆  monthly release 2

  file content rather than as plain links, MemberSpace reports which
  members opened which file. On a membership this size that data is

    /* ===== FILTER, SORT, RENDER ===== */
    function matches(item) {
      if (activeCat === 'new' && !item.isNew) return false;
      if (activeCat !== 'all' && activeCat !== 'new' && item.cat !== activeCat) return false;
      if (!query) return true;
      const sys = SYSTEMS[item.system];
      const hay = (item.title + ' ' + item.blurb + ' ' + (sys ? sys.label : '')).toLowerCase();
      return query.split(/\s+/).every(function (word) { return hay.indexOf(word) !== -1; });
    }

    function sorted(list) {
      const copy = list.slice();
      if (sortBy === 'az') return copy.sort(function (a, b) { return a.title.localeCompare(b.title); });
      if (sortBy === 'short') return copy.sort(function (a, b) { return a.pages - b.pages || a.title.localeCompare(b.title); });
      if (sortBy === 'new') return copy.sort(function (a, b) { return (b.isNew ? 1 : 0) - (a.isNew ? 1 : 0); });
      return copy; /* by system, the order they are written in the array */
    }

      /* text set as text nodes, never innerHTML, so an apostrophe in a

      el.querySelector('h3').textContent = item.title;
      el.querySelector('.blurb').textContent = item.blurb;
      el.querySelector('.get').href = item.link || (FILE_BASE + item.file);

      return el;
    }

    function render() {
      const list = sorted(ITEMS.filter(matches));
      grid.innerHTML = '';
      list.forEach(function (item) { grid.appendChild(cardFor(item)); });

      empty.classList.toggle('show', list.length === 0);
      grid.style.display = list.length === 0 ? 'none' : '';

      const total = ITEMS.length;
      let line;
      if (list.length === total) line = 'Showing all <b>' + total + '</b> pages';
      else if (list.length === 0) line = 'No pages match';
      else line = 'Showing <b>' + list.length + '</b> of ' + total + ' pages';
      if (query) line += ' for &ldquo;' + query.replace(/[<>&]/g, '') + '&rdquo;';
      resultLine.innerHTML = line;
    }
