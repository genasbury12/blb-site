<!--
  ============================================================
  PURPOSEFUL PENNIES  ·  CONTACT (fully linked + fillable)
  URL: /contact
  Draft: https://purposefulpenniesdraft.showit.site/contact
  ============================================================

  HOW THE FORM DELIVERS MAIL  ·  READ THIS FIRST
  ------------------------------------------------------------
  A static HTML page cannot send email on its own. There is no
  server here to hand the message to. Every contact form on the
  internet does one of two things, and this page supports both.

  OPTION A  ·  RELAY SERVICE  (recommended, do this)
    Sign up for a free form relay (Web3Forms, Formspree, Basin
    all work the same way). It gives you a key or a URL. Paste
    it into FORM_ENDPOINT in the JS config below. Messages then
    arrive in Brooke's Gmail inbox as normal email, with no
    visitor needing an email client installed. Confirm the
    current free-tier submission limit when you sign up.

    Web3Forms shape:  FORM_ENDPOINT = "https://api.web3forms.com/submit"
                      ACCESS_KEY    = "your-key-here"
    Formspree shape:  FORM_ENDPOINT = "https://formspree.io/f/yourFormId"
                      ACCESS_KEY    = ""

  OPTION B  ·  MAILTO FALLBACK  (already working, zero setup)
    Leave FORM_ENDPOINT empty. The form composes the message and
    opens the visitor's own mail app addressed to Brooke, with
    subject and body pre-filled. Works today with no account.
    Downside: it depends on the visitor having mail set up, and
    you cannot track submissions. Fine for launch, swap to
    Option A before traffic arrives.

  OPTION C  ·  SHOWIT NATIVE FORM
    If you would rather use Showit's own form element, keep this
    page's layout and copy and swap the .contact-form block for
    the Showit embed. Set its notification address to the Gmail.

  SPAM PROTECTION
    A honeypot field named "_gotcha" is included and visually
    hidden. Bots fill it, humans never see it. Submissions with
    it filled are silently dropped. If spam still gets through,
    add the relay service's own captcha.

  EMAIL ADDRESS EXPOSURE
    The visible address is assembled by JS from two halves so it
    is not sitting in the raw HTML as plain text. This stops
    naive scrapers, not determined ones. See the note in the
    delivery message about moving to a branded forwarding
    address.

  No empty-anchor placeholders. No em dashes or en dashes.
  ============================================================
-->

  /* =========================================================
     FORM DELIVERY CONFIG  ·  the only part you need to touch
     =========================================================

     FORM_ENDPOINT  Leave "" to use the mailto fallback, which
                    works right now with no signup. Paste your
                    relay URL here to have messages arrive in
                    the Gmail inbox as normal email.
     ACCESS_KEY     Only used by Web3Forms style relays. Leave
                    "" for Formspree style endpoints.
     ========================================================= */
  var FORM_ENDPOINT = "";
  var ACCESS_KEY    = "";

  /* Recipient, split so it is not sitting in the raw HTML in one piece */
  var MAIL_USER = "watrousbrooke";
  var MAIL_HOST = "gmail.com";
  var RECIPIENT = MAIL_USER + "@" + MAIL_HOST;

  /* ---------- render the visible address ---------- */
  (function () {
    var slot = document.getElementById('emailSlot');
    if (!slot) return;
    var a = document.createElement('a');
    a.href = 'mailto:' + RECIPIENT;
    a.textContent = RECIPIENT;
    slot.textContent = '';
    slot.appendChild(a);
  })();

  /* ---------- mobile nav ---------- */
  (function () {
    var toggle = document.getElementById('navToggle');
    var links = document.getElementById('navLinks');
    if (!toggle || !links) return;
    toggle.addEventListener('click', function () {
      var open = links.classList.toggle('open');
      toggle.classList.toggle('open', open);
      toggle.setAttribute('aria-expanded', open ? 'true' : 'false');
      toggle.setAttribute('aria-label', open ? 'Close menu' : 'Open menu');
    });
    links.querySelectorAll('a').forEach(function (a) {
      a.addEventListener('click', function () {
        links.classList.remove('open');
        toggle.classList.remove('open');
        toggle.setAttribute('aria-expanded', 'false');
      });
    });
  })();

  /* ---------- fade in ---------- */
  (function () {
    var els = document.querySelectorAll('.fade-in');
    var obs = new IntersectionObserver(function (entries) {
      entries.forEach(function (e) {
        if (e.isIntersecting) { e.target.classList.add('visible'); obs.unobserve(e.target); }
      });
    }, { threshold: 0.1, rootMargin: '0px 0px -40px 0px' });
    els.forEach(function (el) { obs.observe(el); });
  })();

  /* ---------- character count ---------- */
  (function () {
    var ta = document.getElementById('message');
    var out = document.getElementById('charCount');
    if (!ta || !out) return;
    ta.addEventListener('input', function () {
      var n = ta.value.length;
      out.textContent = n === 1 ? '1 character' : n.toLocaleString() + ' characters';
    });
  })();

  /* ---------- FAQ: one open at a time ---------- */
  (function () {
    var items = document.querySelectorAll('.faq-item');
    items.forEach(function (item) {
      item.addEventListener('toggle', function () {
        if (!item.open) return;
        items.forEach(function (other) { if (other !== item) other.open = false; });
      });
    });
  })();

  /* ---------- validation + submit ---------- */
  (function () {
    var form    = document.getElementById('contactForm');
    var fields  = document.getElementById('formFields');
    var success = document.getElementById('formSuccess');
    var status  = document.getElementById('formStatus');
    var btn     = document.getElementById('submitBtn');
    if (!form) return;

    var checks = [
      { wrap: 'f-name',    input: 'name',    test: function (v) { return v.trim().length > 0; } },
      { wrap: 'f-email',   input: 'email',   test: function (v) { return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v.trim()); } },
      { wrap: 'f-topic',   input: 'topic',   test: function (v) { return v !== ''; } },
      { wrap: 'f-message', input: 'message', test: function (v) { return v.trim().length > 0; } }
    ];

    /* clear the error state as soon as they start fixing it */
    checks.forEach(function (c) {
      var el = document.getElementById(c.input);
      var wrap = document.getElementById(c.wrap);
      ['input', 'change'].forEach(function (evt) {
        el.addEventListener(evt, function () {
          if (c.test(el.value)) wrap.classList.remove('invalid');
        });
      });
    });

    function validate() {
      var firstBad = null;
      checks.forEach(function (c) {
        var el = document.getElementById(c.input);
        var wrap = document.getElementById(c.wrap);
        if (c.test(el.value)) {
          wrap.classList.remove('invalid');
        } else {
          wrap.classList.add('invalid');
          if (!firstBad) firstBad = el;
        }
      });
      return firstBad;
    }

    function showSuccess() {
      fields.style.display = 'none';
      success.classList.add('show');
      form.scrollIntoView({ behavior: 'smooth', block: 'center' });
    }

    function buildMailto(d) {
      var subject = 'Purposeful Pennies: ' + d.topic;
      var body =
        'From: ' + d.name + '\n' +
        'Email: ' + d.email + '\n' +
        'Topic: ' + d.topic + '\n\n' +
        d.message + '\n';
      return 'mailto:' + RECIPIENT +
             '?subject=' + encodeURIComponent(subject) +
             '&body=' + encodeURIComponent(body);
    }

    form.addEventListener('submit', function (e) {
      e.preventDefault();
      status.textContent = '';
      status.className = 'form-status';

      /* honeypot: a bot filled the hidden field, pretend everything is fine */
      if (document.getElementById('_gotcha').value !== '') { showSuccess(); return; }

      var bad = validate();
      if (bad) {
        status.className = 'form-status error';
        status.textContent = 'Just a couple of things to fix above.';
        bad.focus();
        return;
      }

      var d = {
        name:    document.getElementById('name').value.trim(),
        email:   document.getElementById('email').value.trim(),
        topic:   document.getElementById('topic').value,
        message: document.getElementById('message').value.trim()
      };

      /* ---- Option B: no relay configured, hand off to their mail app ---- */
      if (!FORM_ENDPOINT) {
        window.location.href = buildMailto(d);
        status.textContent = 'Opening your email app. If nothing happens, the address is in the box on the left.';
        setTimeout(showSuccess, 1200);
        return;
      }

      /* ---- Option A: post to the relay service ---- */
      btn.disabled = true;
      btn.textContent = 'Sending' + '\u2026';
      status.textContent = 'Sending your message.';

      var payload = {
        name: d.name,
        email: d.email,
        topic: d.topic,
        message: d.message,
        subject: 'Purposeful Pennies: ' + d.topic,
        from_name: 'Purposeful Pennies contact form'
      };
      if (ACCESS_KEY) payload.access_key = ACCESS_KEY;

      fetch(FORM_ENDPOINT, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
        body: JSON.stringify(payload)
      })
        .then(function (res) {
          if (!res.ok) throw new Error('Relay responded ' + res.status);
          showSuccess();
        })
        .catch(function () {
          btn.disabled = false;
          btn.textContent = 'Send it to Brooke';
          status.className = 'form-status error';
          status.innerHTML = 'Something went wrong on my end. Please email me directly at ' +
            '<a href="mailto:' + RECIPIENT + '">' + RECIPIENT + '</a> and I will get it.';
        });
    });

    document.getElementById('sendAnother').addEventListener('click', function () {
      form.reset();
      document.getElementById('charCount').textContent = '0 characters';
      checks.forEach(function (c) { document.getElementById(c.wrap).classList.remove('invalid'); });
      success.classList.remove('show');
      fields.style.display = 'block';
      btn.disabled = false;
      btn.textContent = 'Send it to Brooke';
      status.textContent = '';
      status.className = 'form-status';
      document.getElementById('name').focus();
    });
  })();
