function applyFilter(f) {
  var shown = 0;
  document.querySelectorAll('.card').forEach(function (c) {
    var match = (f === 'all' || c.dataset.cat === f);
    c.classList.toggle('hide', !match);
    if (match) shown++;
  });
  document.querySelectorAll('.chip').forEach(function (c) {
    c.classList.toggle('on', c.dataset.filter === f);
  });
  document.getElementById('count').textContent = (f === 'all')
    ? 'Showing all ' + shown + ' resources'
    : 'Showing ' + shown + ' ' + LABELS[f];
}

/* ---------- open the library ---------- */
function openLibrary() {
  document.getElementById('gate').style.display = 'none';
  var lib = document.getElementById('library');
  lib.classList.add('open');
  render();
  document.querySelectorAll('.chip').forEach(function (c) {
    c.addEventListener('click', function () { applyFilter(c.dataset.filter); });
  });
  window.scrollTo(0, 0);
}

/* ---------- one click entry from the thank-you page ----------
   /library?u=<hash> unlocks without typing. The hash is already in
   this file's source, so this adds no new exposure. -------------- */
(function autoUnlock() {
  var u = '';
  try { u = new URLSearchParams(window.location.search).get('u') || ''; } catch (e) { return; }
  if (u && String(PASS_HASH) === u) {
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', openLibrary);
    } else { openLibrary(); }
  }
})();

/* ---------- the gate ---------- */
document.getElementById('gateForm').addEventListener('submit', function (e) {
  e.preventDefault();
  var gate = document.getElementById('gate');
  var val = document.getElementById('pw').value.trim().toLowerCase();
  if (h(val) === PASS_HASH) {
    openLibrary();
  } else {
    gate.classList.add('error');
    document.getElementById('pw').select();
  }
});
document.getElementById('pw').addEventListener('input', function () {
  document.getElementById('gate').classList.remove('error');
});
